import React from 'react';

let injected = false;
function inject() {
  if (injected || typeof document === 'undefined') return;
  injected = true;
  const css = `
.zk-tip{position:relative;display:inline-flex;}
.zk-tip__bubble{position:absolute;z-index:50;font-family:var(--font-sans);font-size:var(--text-xs);font-weight:var(--fw-medium);
  color:var(--ivory);background:var(--espresso);border:1px solid var(--border-dark);border-radius:var(--radius-sm);
  padding:6px 10px;white-space:nowrap;box-shadow:var(--shadow-md);pointer-events:none;opacity:0;
  transition:opacity var(--duration-fast) var(--ease-standard),transform var(--duration-fast) var(--ease-out);}
.zk-tip__bubble::after{content:"";position:absolute;width:8px;height:8px;background:var(--espresso);
  border:1px solid var(--border-dark);transform:rotate(45deg);}
.zk-tip--open .zk-tip__bubble{opacity:1;}
.zk-tip__bubble--top{bottom:calc(100% + 8px);left:50%;transform:translateX(-50%) translateY(4px);}
.zk-tip--open .zk-tip__bubble--top{transform:translateX(-50%) translateY(0);}
.zk-tip__bubble--top::after{bottom:-5px;left:50%;margin-left:-4px;border-top:none;border-left:none;}
.zk-tip__bubble--bottom{top:calc(100% + 8px);left:50%;transform:translateX(-50%) translateY(-4px);}
.zk-tip--open .zk-tip__bubble--bottom{transform:translateX(-50%) translateY(0);}
.zk-tip__bubble--bottom::after{top:-5px;left:50%;margin-left:-4px;border-bottom:none;border-right:none;}
`;
  const el = document.createElement('style');
  el.setAttribute('data-zk', 'tooltip');
  el.textContent = css;
  document.head.appendChild(el);
}

export function Tooltip({ label, placement = 'top', className = '', children }) {
  inject();
  const [open, setOpen] = React.useState(false);
  return (
    <span className={['zk-tip', open ? 'zk-tip--open' : '', className].filter(Boolean).join(' ')}
      onMouseEnter={() => setOpen(true)} onMouseLeave={() => setOpen(false)}
      onFocus={() => setOpen(true)} onBlur={() => setOpen(false)}>
      {children}
      <span role="tooltip" className={`zk-tip__bubble zk-tip__bubble--${placement}`}>{label}</span>
    </span>
  );
}
