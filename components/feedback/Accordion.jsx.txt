import React from 'react';

let injected = false;
function inject() {
  if (injected || typeof document === 'undefined') return;
  injected = true;
  const css = `
.zk-acc{border-top:1px solid var(--border-soft);}
.zk-acc__item{border-bottom:1px solid var(--border-soft);}
.zk-acc__head{width:100%;display:flex;align-items:center;justify-content:space-between;gap:var(--space-4);
  font-family:var(--font-display);font-size:var(--text-display-sm);font-weight:var(--fw-display-regular);color:var(--text-strong);
  letter-spacing:var(--tracking-display);text-align:left;background:none;border:none;cursor:pointer;padding:var(--space-5) 0;line-height:1.2;}
.zk-acc__head:hover{color:var(--gold-deep);}
.zk-acc__head:focus-visible{outline:2px solid var(--focus-ring);outline-offset:2px;border-radius:2px;}
.zk-acc__icon{flex:0 0 auto;width:24px;height:24px;color:var(--gold-deep);transition:transform var(--duration-base) var(--ease-out);}
.zk-acc__icon svg{width:100%;height:100%;stroke-width:1.5;}
.zk-acc__item--open .zk-acc__icon{transform:rotate(45deg);}
.zk-acc__panel{overflow:hidden;transition:height var(--duration-base) var(--ease-standard);}
.zk-acc__body{font-family:var(--font-sans);font-size:var(--text-md);color:var(--text-muted);line-height:var(--lh-relaxed);
  padding:0 0 var(--space-5);max-width:64ch;}
`;
  const el = document.createElement('style');
  el.setAttribute('data-zk', 'accordion');
  el.textContent = css;
  document.head.appendChild(el);
}

const Plus = () => (<svg viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeLinecap="round"><path d="M12 5v14M5 12h14"/></svg>);

function AccItem({ item, isOpen, onToggle }) {
  const ref = React.useRef(null);
  const [height, setHeight] = React.useState(0);
  React.useEffect(() => { if (ref.current) setHeight(ref.current.scrollHeight); }, [item]);
  return (
    <div className={['zk-acc__item', isOpen ? 'zk-acc__item--open' : ''].filter(Boolean).join(' ')}>
      <button className="zk-acc__head" aria-expanded={isOpen} onClick={onToggle}>
        <span>{item.title}</span>
        <span className="zk-acc__icon"><Plus /></span>
      </button>
      <div className="zk-acc__panel" style={{ height: isOpen ? height : 0 }}>
        <div className="zk-acc__body" ref={ref}>{item.content}</div>
      </div>
    </div>
  );
}

export function Accordion({ items = [], allowMultiple = false, defaultOpen = [], className = '' }) {
  inject();
  const [open, setOpen] = React.useState(new Set(defaultOpen));
  const toggle = (idx) => setOpen((prev) => {
    const next = new Set(allowMultiple ? prev : []);
    if (prev.has(idx)) next.delete(idx); else next.add(idx);
    return next;
  });
  return (
    <div className={['zk-acc', className].filter(Boolean).join(' ')}>
      {items.map((it, i) => <AccItem key={i} item={it} isOpen={open.has(i)} onToggle={() => toggle(i)} />)}
    </div>
  );
}
